## Global Constraints

- **An explicit value is never changed.** A call that passes `model` keeps it (existing behaviour, reason `explicit`). A call that passes `effort` keeps its effort.
- **The role floor stays:** `ROLE_FLOOR_RE = re.compile(r"review|architect", re.I)`. A matching type is never emitted below sonnet.
- **One threshold for both questions**, read from `<git common dir>/jev/config.json` `threshold`. Defaults: `THRESHOLD_DEFAULT = 0.65`, and `CONFIG_ON["threshold"] = 0.65`.
- **Effort values are the router's `EFFORTS = ("low", "medium", "high", "xhigh")`.** The Agent tool accepts these as its `effort` input.
- **The router never fails a spawn.** It exits 0 always; the existing `except Exception` floor path stays.
- **Never print, log or store** the key, the prompt or env values. An event holds the fields in `EVENT_FIELDS` only.
- **Jev unit tests:** run `python3 -B -m unittest discover -s skills/jev/tests` from the repo root (no `-t`: the tests dir has no `__init__.py`). The hook suite already runs them in its v4.4.0 **J-PY** block (`scripts/test-hooks.sh`), which also pins the test count in `JPY_WANT`. A task that adds or removes Jev tests updates `JPY_WANT`.
- **All files are LF.** KISS. Every commit message ends with the session trailer lines given in the dispatch.

```python
THRESHOLD_DEFAULT = 0.65


def decide(kind, default, subagent_type, answer, threshold):
    """Bound Jev's model answer. -> (model to emit or None, applied, reason).

    A floor spawn always gets a model (R-1): the floor itself when Jev's choice
    is not applied. A typed spawn gets one only for an applied move. v5.1: any
    step on haiku < sonnet < opus < fable; review/architect never below sonnet.
    """
    fallback = default if kind == "floor" else None
    fam = family(default)
    if fam is None:
        return fallback, False, "pinned"
    if not answer:
        return fallback, False, "bad-response"
    choice, conf = answer["choice"], answer["confidence"]
    if conf < threshold:
        return fallback, False, "low-confidence"
    if choice == fam:
        return fallback, False, "kept"
    if ROLE_FLOOR_RE.search(subagent_type or "") and ORDER.index(choice) < ORDER.index("sonnet"):
        return fallback, False, "role-floor"
    return choice, True, "applied"


def decide_effort(agent_effort, explicit_effort, answer, threshold):
    """v5.1: Jev's effort, only when confident and different. -> (effort to emit or None, reason)."""
    if explicit_effort:
        return None, "explicit"
    if not answer:
        return None, "bad-response"
    if answer["confidence"] < threshold:
        return None, "low-confidence"
    if answer["choice"] == agent_effort:
        return None, "kept"
    return answer["choice"], "applied"


def emit(tool_input, model, effort=None):
    """The hook's stdout: the WHOLE original tool_input with `model` and/or `effort` set
    (updatedInput REPLACES it). b"" when there is nothing to set or it cannot be strict JSON."""
    if model is None and effort is None:
        return b""
    try:
        ti = dict(tool_input)
        if model is not None:
            ti["model"] = model
        if effort is not None:
            ti["effort"] = effort
        out = json.dumps({"hookSpecificOutput": {"hookEventName": "PreToolUse", "updatedInput": ti}},
                         ensure_ascii=True, allow_nan=False)
    except (TypeError, ValueError):
        return b""
    return out.encode("utf-8")
```

"""scripts/jev-tally.py: the cross-machine trial tally (v5.1)."""
import importlib.util
import json
import os
import shutil
import sys
import tempfile
import unittest

sys.dont_write_bytecode = True
HERE = os.path.dirname(os.path.abspath(__file__))
SCRIPT = os.path.join(HERE, "..", "..", "..", "scripts", "jev-tally.py")


def load():
    spec = importlib.util.spec_from_file_location("jev_tally", SCRIPT)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod


class TallyTests(unittest.TestCase):
    def setUp(self):
        self.root = tempfile.mkdtemp(prefix="jev-tally-")
        self.addCleanup(shutil.rmtree, self.root, True)
        self.t = load()

    def ev(self, repo, name, **kw):
        d = os.path.join(self.root, repo, ".git", "jev", "events")
        os.makedirs(d, exist_ok=True)
        base = {"ts": "2026-10-20T10:00:00.000Z", "subagent_type": "coder", "kind": "own", "default": "sonnet",
                "choice": "opus", "confidence": 0.9, "applied": True, "reason": "applied", "emitted": "opus",
                "latency_s": 0.3, "effort_emitted": None}
        base.update(kw)
        with open(os.path.join(d, name), "w") as fh:
            json.dump(base, fh)

    def test_counts_applied_up_and_reasons(self):
        self.ev("a", "1.json")
        self.ev("a", "2.json", applied=False, reason="low-confidence", emitted=None, confidence=0.5)
        self.ev("b", "3.json", default="opus", emitted="sonnet", effort_emitted="high")
        r = self.t.tally([self.root], None)
        self.assertEqual((r["events"], r["repos"], r["applied"], r["up"], r["down"]), (3, 2, 2, 1, 1))

        self.assertEqual((r["events"], r["skipped_files"]), (1, 1))

    def test_empty_root_is_zero_not_a_crash(self):
